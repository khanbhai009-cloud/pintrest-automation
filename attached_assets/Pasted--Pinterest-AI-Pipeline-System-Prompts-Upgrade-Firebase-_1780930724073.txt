```python
import os
import time
import requests

FIREBASE_URL = os.getenv("FIREBASE_URL")       # https://your-project-default-rtdb.firebaseio.com
FIREBASE_SECRET = os.getenv("FIREBASE_SECRET") # Optional: Database secret for auth

def _fb_get(path: str):
    url = f"{FIREBASE_URL}/{path}.json"
    if FIREBASE_SECRET:
        url += f"?auth={FIREBASE_SECRET}"
    try:
        r = requests.get(url, timeout=5)
        return r.json() if r.ok else None
    except Exception as e:
        print(f"[Firebase] GET error {path}: {e}")
        return None

def get_boards(account: str) -> dict:
    """Get all active boards for account, sorted by priority"""
    data = _fb_get(f"boards/{account}")
    if not data:
        return {}
    active = {k: v for k, v in data.items() if v.get("active", True)}
    return dict(sorted(active.items(), key=lambda x: x[1].get("priority", 99)))

def get_active_trends(account: str, niche_key: str) -> list:
    """Get active trend keywords for a niche — returns [] if expired"""
    data = _fb_get(f"trends/{account}/{niche_key}")
    if not data:
        return []
    if data.get("expires_at", 0) < time.time() * 1000:  # expires_at stored in ms
        return []
    return data.get("keywords", [])

def get_all_active_trends(account: str) -> dict:
    """Get all active trend keyword sets for account — {niche_key: [kw1, kw2]}"""
    data = _fb_get(f"trends/{account}")
    if not data:
        return {}
    now = time.time() * 1000
    return {
        niche_key: trend["keywords"]
        for niche_key, trend in data.items()
        if trend.get("expires_at", 0) > now and trend.get("keywords")
    }

def format_boards_for_prompt(boards: dict) -> str:
    """Format boards dict into readable string for AI prompt"""
    if not boards:
        return "No boards configured."
    lines = []
    for niche_key, board in boards.items():
        kws = ", ".join(board.get("niche_keywords", []))
        lines.append(
            f"- [{niche_key}] \"{board.get('board_name', niche_key)}\"\n"
            f"  Board ID: {board.get('board_id', 'MISSING')}\n"
            f"  Description: {board.get('description', 'N/A')}\n"
            f"  Match keywords: {kws}\n"
            f"  Priority: {board.get('priority', 99)}"
        )
    return "\n\n".join(lines)

def format_trends_for_prompt(trends: dict) -> str:
    """Format active trends into readable string for AI prompt"""
    if not trends:
        return "No active trend keywords this week."
    lines = []
    for niche_key, keywords in trends.items():
        lines.append(f"- {niche_key}: {', '.join(keywords[:10])}")
    return "\n".join(lines)
```

```python
def node_firebase_loader(state: MastermindState) -> dict:

    Load boards and active trend keywords from Firebase.
    Runs BEFORE CMO node.

    account = state.get("account", "account_1")

        "boards": boards,
        "trend_keywords": trends,
```

## 📁 Step 5: Update `node_seo.py` — Trend-Keyword-Driven Copy

Replace your existing SEO prompt with this:

```python
# ─── UPDATED SEO SYSTEM PROMPT ───
SEO_SYSTEM_PROMPT = """
You are a Pinterest SEO specialist. Your job: write pin copy that ranks and saves.

━━━ CONTEXT ━━━
Board: {board_name} ({board_niche})
Pin visual: {visual_summary}
Primary keyword: {primary_keyword}
Active trend keywords: {active_keywords_str}

━━━ TITLE RULES (MOST CRITICAL) ━━━
Pinterest indexes first 40 characters most heavily.

FORMULA: [Primary Keyword] — [Hook/Benefit/Number]
GOOD: "Cozy Bedroom Ideas 2025 — 12 Dreamy Looks"
GOOD: "Cottagecore Garden Goals That Will Transform Your Space"  
GOOD: "Kawaii Gaming Setup — Pink Aesthetic Desk Ideas"
BAD:  "Beautiful Room Decoration" (no keyword, no hook)
BAD:  "Amazing Home Decor Ideas You'll Love" (too generic)

Rules:
- Characters 1-40 MUST contain primary_keyword (exact or close variation)
- Total length: 60-100 characters
- Include a number OR power word (Dreamy, Cozy, Ultimate, Perfect, Stunning)
- Never use clickbait — Pinterest audience saves real inspiration

━━━ DESCRIPTION RULES ━━━
- Length: 150-250 words
- Sentence 1: Must contain primary_keyword naturally
- Weave in 3-4 trend keywords throughout (never stuffed back-to-back)
- Include 1 question mid-description (boosts engagement)
- Soft CTA at end: "Save this for later 📌" or "Try this look →"
- Write as a knowledgeable friend, not a salesperson
- No hashtags inside description

━━━ HASHTAG RULES ━━━
Generate exactly 15 hashtags:
- 3 mega (10M+ posts): e.g. #HomeDecor #BedroomIdeas #InteriorDesign
- 5 macro (1-10M posts): e.g. #CozyBedroom #AestheticRoom #BedroomInspo
- 4 micro (100K-1M posts): e.g. #DreamyBedroom2025 #CozyHomeAesthetic
- 3 niche: convert trend keywords to hashtags (remove spaces, title case)

Always include #{primary_keyword_as_hashtag} in micro or niche group.

━━━ ALT TEXT ━━━
One sentence, descriptive, includes primary_keyword naturally.
For Pinterest SEO, this matters. Write like you're describing to someone who can't see.

━━━ OUTPUT FORMAT ━━━
Return JSON only:
{{
  "title": "...",
  "description": "...",
  "hashtags": ["#tag1", "#tag2", ...],
  "alt_text": "..."
}}
"""

def build_seo_prompt(state: MastermindState) -> str:
    keywords = state.get("active_keywords", [])
    primary = state.get("primary_keyword", "")
    
    # Fallback: use board keywords if no trends
    if not primary and state.get("boards"):
        board_niche = state.get("selected_board_niche", "")
        board = state["boards"].get(board_niche, {})
        board_kws = board.get("niche_keywords", [])
        primary = board_kws[0] if board_kws else "home decor ideas"
        keywords = board_kws[:5]
    
    return SEO_SYSTEM_PROMPT.format(
        board_name=state.get("selected_board_name", "Home Decor"),
        board_niche=state.get("selected_board_niche", "home_decor"),
        visual_summary=state.get("visual_prompt", "")[:200],
        primary_keyword=primary,
        active_keywords_str=", ".join(keywords) if keywords else "none this week",
        primary_keyword_as_hashtag="".join(w.capitalize() for w in primary.split()) if primary else "HomeDecor",
    )
```

```python
from langgraph.graph import StateGraph, END
from nodes.node_firebase import node_firebase_loader
from nodes.node_cmo import node_cmo
from nodes.node_seo import node_seo
from nodes.node_execute import node_execute

def build_graph():
    g = StateGraph(MastermindState)
    
    # Add nodes
    g.add_node("firebase_loader", node_firebase_loader)  # NEW — first node
    g.add_node("cmo", node_cmo)
    g.add_node("seo", node_seo)
    g.add_node("execute", node_execute)
    
    # Edges — firebase_loader runs BEFORE cmo
    g.set_entry_point("firebase_loader")
    g.add_edge("firebase_loader", "cmo")
    g.add_edge("cmo", "seo")
    g.add_edge("seo", "execute")
    g.add_edge("execute", END)
    
    return g.compile()
```
